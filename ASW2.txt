! ============================================================
! ASW2 - Access switch 2 (Layer 2)
! Sanitized for public release (secrets removed)
! After pasting: generate the RSA key once (not stored in running-config):
!   crypto key generate rsa modulus 1024
! ============================================================
hostname ASW2
!
enable secret <REDACTED>
username admin privilege 15 secret <REDACTED>
!
ip domain-name darlylab.local
ip ssh version 2
!
spanning-tree mode rapid-pvst
spanning-tree extend system-id
!
! Management PC (VLAN 99)
interface FastEthernet0/1
 switchport access vlan 99
 switchport mode access
 spanning-tree portfast
 spanning-tree bpduguard enable
!
! VLAN 20 host behind IP phone (voice VLAN 120)
interface FastEthernet0/2
 switchport access vlan 20
 switchport mode access
 switchport voice vlan 120
 spanning-tree portfast
 spanning-tree bpduguard enable
!
! Trunks to DSW1 and DSW2
interface range FastEthernet0/3 - 4
 switchport trunk native vlan 150
 switchport trunk allowed vlan 10,20,30,99,120,150
 switchport mode trunk
!
! DHCP server (VLAN 30)
interface FastEthernet0/5
 switchport access vlan 30
 switchport mode access
 spanning-tree portfast
 spanning-tree bpduguard enable
!
! In-band management
interface Vlan99
 description Management SVI
 ip address 192.168.99.12 255.255.255.0
 no shutdown
!
ip default-gateway 192.168.99.1
!
access-list 99 permit 192.168.99.0 0.0.0.255
access-list 99 deny any
!
line con 0
 exec-timeout 30 0
 logging synchronous
 login local
!
line vty 0 15
 access-class 99 in
 login local
 transport input ssh
 privilege level 15
!
end
